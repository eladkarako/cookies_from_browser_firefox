1//! Load, sort, filter, and output Firefox or Netscape cookie files.
2//!
3//! The input can be either:
4//!
5//! - A Firefox `cookies.sqlite` database.
6//! - A Netscape-format cookie text file.
7//!
8//! The input type is detected from the first 16 bytes. SQLite databases
9//! begin with `SQLite format 3\0`; all other files are treated as Netscape
10//! cookie files.
11//!
12//! Output is always written in Netscape cookie-file format.
13
14use anyhow::{Context, Result};

18    fs::File,
19    io::{self, BufRead, BufReader, BufWriter, Read, Write},
20    path::{Path, PathBuf},

23use tokio::task::JoinHandle;

25/// Command-line arguments accepted by the program.
26#[derive(Parser, Debug)]
27#[command(
28    author,
29    version,
30    about = env!("CARGO_PKG_DESCRIPTION"),
31)]
32struct Args {
33    /// Input path.
34    ///
35    /// This can be a Firefox cookies.sqlite database or a Netscape
36    /// cookie text file.
37    #[arg(long)]
38    path: PathBuf,
39
40    /// Include hosts containing this case-insensitive partial match.
41    ///
42    /// This option may be specified multiple times. Multiple include
43    /// values use OR semantics.
44    ///
45    /// For example:
46    ///
47    ///     --include google --include mozilla
48    ///
49    /// keeps a cookie if its host contains either "google" or "mozilla".
50    #[arg(long = "include", value_name = "TEXT")]
51    includes: Vec<String>,
52
53    /// Exclude hosts containing this case-insensitive partial match.
54    ///
55    /// This option may be specified multiple times. Exclude filters take
56    /// precedence over include filters.
57    ///
58    /// For example:
59    ///
60    ///     --exclude ads --exclude tracker
61    ///
62    /// removes a cookie if its host contains either "ads" or "tracker".
63    #[arg(long = "exclude", value_name = "TEXT")]
64    excludes: Vec<String>,
65}
66
67/// One cookie represented independently of its input format.
68///
69/// The Netscape file format stores `HttpOnly` as part of the domain field
70/// using the `#HttpOnly_` prefix. Internally, this prefix is removed and
71/// represented by the separate `http_only` field.
72#[derive(Debug, Clone)]
73struct Cookie {
74    /// Cookie domain, without the Netscape `#HttpOnly_` prefix.
75    host: String,
76
77    /// Cookie path.
78    path: String,
79
80    /// Whether the cookie is restricted to HTTPS.
81    secure: bool,
82
83    /// Whether JavaScript cannot access the cookie.
84    http_only: bool,
85
86    /// Expiration time as a Unix timestamp in seconds.
87    expiry: i64,
88
89    /// Cookie name.
90    name: String,
91
92    /// Cookie value.
93    value: String,
94}

96/// The supported input file types.
97#[derive(Debug, Clone, Copy, PartialEq, Eq)]
98enum InputType {
99    /// A Firefox SQLite cookie database.
100    Sqlite,
101
102    /// A Netscape-format cookie text file.
103    Netscape,
104}

106/// Program entry point.
107///
108/// The function loads the input, sorts it, applies filters, serializes the
109/// final result into an in-memory byte buffer, and writes that buffer to
110/// standard output.
111#[tokio::main(flavor = "multi_thread")]
112async fn main() -> Result<()> {
113    // Parse command-line arguments.
114    let args = Args::parse();
115
116    // Check that the input path exists and is a regular file.
117    if !args.path.is_file() {
118        anyhow::bail!(
119            "input file does not exist: {}",
120            args.path.display()
121        );
122    }
123
124    // Detect whether the input is SQLite or Netscape text.
125    let input_type = detect_input_type(&args.path)?;
126
127    // Load all cookies into one common in-memory representation.
128    let mut cookies = match input_type {
129        // SQLite reading uses multiple blocking worker tasks.
130        InputType::Sqlite => load_sqlite_cookies(&args.path).await?,
131
132        // Netscape input is already text and does not require SQLite.
133        InputType::Netscape => load_netscape_cookies(&args.path)?,
134    };
135
136    // Report the number of records loaded before filtering.
137    eprintln!("Loaded cookies: {}", cookies.len());
138
139    // Sort before filtering, as requested.
140    sort_cookies(&mut cookies);
141
142    // Apply include and exclude filters to the sorted records.
143    filter_cookies(&mut cookies, &args.includes, &args.excludes);
144
145    // Report the number of records remaining after filtering.
146    eprintln!("Output cookies: {}", cookies.len());
147
148    // Serialize all records into one cached output byte buffer.
149    let output = serialize_netscape(&cookies);
150
151    // Buffer the pipe output to reduce the number of write operations.
152    let stdout = io::stdout();
153    let mut writer = BufWriter::new(stdout.lock());
154
155    // Write the complete cached output to stdout.
156    writer.write_all(&output).context("failed to write output")?;
157
158    // Flush the buffered writer so all bytes reach the pipe.
159    writer.flush().context("failed to flush output")?;
160
161    Ok(())
162}

164/// Detects whether `path` contains SQLite or Netscape-format data.
165///
166/// SQLite databases have a fixed 16-byte header:
167///
168/// ```text
169/// SQLite format 3\0
170/// ```
171///
172/// Any file that does not have this header is treated as a Netscape cookie
173/// file.
174fn detect_input_type(path: &Path) -> Result<InputType> {
175    // Open the file for reading its header.
176    let mut file = File::open(path).with_context(|| {
177        format!("failed to open {}", path.display())
178    })?;
179
180    // The SQLite signature is exactly 16 bytes long.
181    let mut header = [0u8; 16];
182
183    // Read up to 16 bytes from the beginning of the file.
184    let bytes_read = file.read(&mut header).with_context(|| {
185        format!("failed to read {}", path.display())
186    })?;
187
188    // Compare the complete SQLite signature.
189    if bytes_read == 16 && &header == b"SQLite format 3\0" {
190        Ok(InputType::Sqlite)
191    } else {
192        // Non-SQLite input is interpreted as Netscape text.
193        Ok(InputType::Netscape)
194    }
195}

197/// Loads cookies from a Firefox SQLite database.
198///
199/// The database is divided into chunks and read by blocking worker tasks.
200/// Every worker returns structured `Cookie` values. The calling task waits
201/// for every worker and combines their results before sorting.
202async fn load_sqlite_cookies(path: &Path) -> Result<Vec<Cookie>> {
203    // Count the rows before dividing the database into worker chunks.
204    let row_count = {
205        // Move an owned path into the blocking task.
206        let path = path.to_path_buf();
207
208        // SQLite operations are blocking, so run the count outside the
209        // asynchronous executor thread.
210        tokio::task::spawn_blocking(move || count_rows(&path))
211            .await
212            .context("row-count task panicked")??
213    };
214
215    // Return an empty result for an empty database.
216    if row_count == 0 {
217        return Ok(Vec::new());
218    }
219
220    // Determine how many blocking workers can reasonably be used.
221    let available_workers = std::thread::available_parallelism()
222        .map(|value| value.get())
223        .unwrap_or(1);
224
225    // Do not create more workers than rows.
226    let worker_count = available_workers.min(row_count);
227
228    // Divide the rows approximately evenly between workers.
229    let rows_per_worker = row_count.div_ceil(worker_count);
230
231    // Report SQLite processing details on stderr.
232    eprintln!("SQLite rows: {row_count}");
233    eprintln!("SQLite workers: {worker_count}");
234
235    // Share the database path between worker tasks.
236    let database_path = Arc::new(path.to_path_buf());
237
238    // Each worker returns a vector of structured cookies.
239    let mut handles: Vec<JoinHandle<Result<Vec<Cookie>>>> =
240        Vec::with_capacity(worker_count);
241
242    // Create one task for each database chunk.
243    for worker_index in 0..worker_count {
244        // Calculate the first row assigned to this worker.
245        let offset = worker_index * rows_per_worker;
246
247        // Calculate how many rows remain after this worker's offset.
248        let remaining = row_count - offset;
249
250        // Limit this worker to its normal chunk size or the remaining rows.
251        let limit = remaining.min(rows_per_worker);
252
253        // Avoid starting an empty worker.
254        if limit == 0 {
255            continue;
256        }
257
258        // Clone the shared path reference for this worker.
259        let database_path = Arc::clone(&database_path);
260
261        // Spawn the blocking SQLite operation.
262        let handle = tokio::task::spawn_blocking(move || {
263            read_sqlite_chunk(&database_path, offset, limit)
264        });
265
266        // Store the handle so it can be joined below.
267        handles.push(handle);
268    }
269
270    // Allocate enough room for the complete result.
271    let mut cookies = Vec::with_capacity(row_count);
272
273    // Wait for every worker and merge all worker results.
274    for handle in handles {
275        // The first `?` handles task panics; the second handles the
276        // Result returned by the worker itself.
277        let worker_cookies = handle
278            .await
279            .context("SQLite reader task panicked")??;
280
281        // Append this worker's records to the combined result.
282        cookies.extend(worker_cookies);
283    }
284
285    // Sorting is intentionally performed after all workers are joined.
286    Ok(cookies)
287}

289/// Opens a SQLite database in read-only mode.
290fn open_readonly(path: &Path) -> Result<Connection> {
291    // Use SQLite's read-only open flag.
292    let flags = OpenFlags::SQLITE_OPEN_READ_ONLY;
293
294    // Open the database and add its path to any error message.
295    Connection::open_with_flags(path, flags).with_context(|| {
296        format!("failed to open {}", path.display())
297    })
298}

300/// Counts records in the Firefox `moz_cookies` table.
301fn count_rows(path: &Path) -> Result<usize> {
302    // Open the database without allowing writes.
303    let connection = open_readonly(path)?;
304
305    // Query the number of cookie rows.
306    let count: i64 = connection.query_row(
307        "SELECT COUNT(*) FROM moz_cookies",
308        [],
309        |row| row.get(0),
310    )?;
311
312    // Convert SQLite's signed integer to Rust's platform-sized integer.
313    usize::try_from(count)
314        .context("SQLite row count does not fit in usize")
315}

317/// Reads one SQLite chunk and converts it into structured cookies.
318fn read_sqlite_chunk(
319    path: &Path,
320    offset: usize,
321    limit: usize,
322) -> Result<Vec<Cookie>> {
323    // Open one independent read-only connection for this worker.
324    let connection = open_readonly(path)?;
325
326    // Select the fields needed for Netscape output.
327    let mut statement = connection.prepare(
328        r#"
329        SELECT
330            host,
331            path,
332            isSecure,
333            isHttpOnly,
334            expiry,
335            name,
336            value
337        FROM moz_cookies
338        LIMIT ?1 OFFSET ?2
339        "#,
340    )?;
341
342    // SQLite bind parameters use signed 64-bit integers.
343    let limit_i64 = i64::try_from(limit)
344        .context("chunk size does not fit in SQLite integer")?;
345
346    // Convert the offset to SQLite's integer type.
347    let offset_i64 = i64::try_from(offset)
348        .context("offset does not fit in SQLite integer")?;
349
350    // Execute the chunk query.
351    let mut rows = statement.query((limit_i64, offset_i64))?;
352
353    // Reserve enough capacity for this worker's expected result.
354    let mut cookies = Vec::with_capacity(limit);
355
356    // Read every row in this worker's chunk.
357    while let Some(row) = rows.next()? {
358        // Read the cookie domain.
359        let host =
360            row.get::<_, Option<String>>(0)?.unwrap_or_default();
361
362        // Use "/" if the database path is NULL.
363        let path = row
364            .get::<_, Option<String>>(1)?
365            .unwrap_or_else(|| "/".to_string());
366
367        // SQLite stores boolean values as integer values.
368        let is_secure = row.get::<_, Option<i64>>(2)?.unwrap_or(0);
369
370        // Read the separate HttpOnly flag.
371        let is_http_only =
372            row.get::<_, Option<i64>>(3)?.unwrap_or(0);
373
374        // Read the expiration timestamp.
375        let expiry = row.get::<_, Option<i64>>(4)?.unwrap_or(0);
376
377        // Read the cookie name.
378        let name =
379            row.get::<_, Option<String>>(5)?.unwrap_or_default();
380
381        // Read the cookie value.
382        let value =
383            row.get::<_, Option<String>>(6)?.unwrap_or_default();
384
385        // Store Secure and HttpOnly independently.
386        cookies.push(Cookie {
387            host,
388            path,
389            secure: is_secure != 0,
390            http_only: is_http_only != 0,
391            expiry,
392            name,
393            value,
394        });
395    }
396
397    // Return this worker's structured records.
398    Ok(cookies)
399}

401/// Loads a Netscape-format cookie file.
402fn load_netscape_cookies(path: &Path) -> Result<Vec<Cookie>> {
403    // Open the text file.
404    let file = File::open(path).with_context(|| {
405        format!("failed to open {}", path.display())
406    })?;
407
408    // Read the file line by line.
409    let reader = BufReader::new(file);
410
411    // Store parsed cookies here.
412    let mut cookies = Vec::new();
413
414    // Process every input line.
415    for (line_index, line_result) in reader.lines().enumerate() {
416        // Convert the zero-based index into a human-readable line number.
417        let line_number = line_index + 1;
418
419        // Read one line and attach its line number to errors.
420        let line = line_result.with_context(|| {
421            format!("failed reading line {line_number}")
422        })?;
423
424        // Remove a Windows carriage return if present.
425        let line = line.trim_end_matches('\r');
426
427        // Ignore empty lines.
428        if line.is_empty() {
429            continue;
430        }
431
432        /*
433         * Ordinary comments begin with '#'.
434         *
435         * HttpOnly cookie records are a special exception because their
436         * domain begins with "#HttpOnly_".
437         */
438        if line.starts_with('#') && !line.starts_with("#HttpOnly_") {
439            continue;
440        }
441
442        // Netscape cookie records must contain exactly seven fields.
443        let fields: Vec<&str> = line.split('\t').collect();
444
445        // Reject malformed records instead of silently corrupting output.
446        if fields.len() != 7 {
447            anyhow::bail!(
448                "invalid Netscape cookie line {line_number}: \
449                 expected 7 tab-separated fields, found {}",
450                fields.len()
451            );
452        }
453
454        // Detect and remove the special HttpOnly domain prefix.
455        let (http_only, host) = if let Some(host) =
456            fields[0].strip_prefix("#HttpOnly_")
457        {
458            (true, host.to_string())
459        } else {
460            (false, fields[0].to_string())
461        };
462
463        // Validate the include-subdomains field.
464        //
465        // The output later derives this field from the domain's leading
466        // dot, but validating the input catches malformed cookie files.
467        let _include_subdomains = parse_bool_field(fields[1])
468            .with_context(|| {
469                format!(
470                    "invalid include_subdomains field on line {line_number}"
471                )
472            })?;
473
474        // Parse the Secure field.
475        let secure =
476            parse_bool_field(fields[3]).with_context(|| {
477                format!("invalid secure field on line {line_number}")
478            })?;
479
480        // Parse the Unix expiration timestamp.
481        let expiry =
482            fields[4].parse::<i64>().with_context(|| {
483                format!(
484                    "invalid expiration field on line {line_number}"
485                )
486            })?;
487
488        // Store the parsed Netscape record.
489        cookies.push(Cookie {
490            host,
491            path: fields[2].to_string(),
492            secure,
493            http_only,
494            expiry,
495            name: fields[5].to_string(),
496            value: fields[6].to_string(),
497        });
498    }
499
500    // Return all parsed records.
501    Ok(cookies)
502}

504/// Parses a Netscape boolean field.
505///
506/// Accepted values are `TRUE`, `FALSE`, `1`, and `0`, without regard to
507/// letter case.
508fn parse_bool_field(value: &str) -> Result<bool> {
509    // Normalize the field before comparing it.
510    match value.trim().to_ascii_lowercase().as_str() {
511        // These values represent true.
512        "true" | "1" => Ok(true),
513
514        // These values represent false.
515        "false" | "0" => Ok(false),
516
517        // Reject unknown values.
518        other => anyhow::bail!("expected TRUE/FALSE, got {other:?}"),
519    }
520}

522/// Sorts cookies by normalized host, path, and name.
523///
524/// A leading dot is ignored while comparing hosts:
525///
526/// ```text
527/// .example.com
528/// example.com
529/// ```
530///
531/// are compared using the same normalized host, `example.com`.
532fn sort_cookies(cookies: &mut [Cookie]) {
533    // Sort records in place without allocating a second vector.
534    cookies.sort_unstable_by(|a, b| {
535        // Remove leading dots only for comparison.
536        let a_host = a.host.trim_start_matches('.');
537        let b_host = b.host.trim_start_matches('.');
538
539        // Apply the requested primary ordering.
540        a_host
541            .cmp(b_host)
542            .then_with(|| a.path.cmp(&b.path))
543            .then_with(|| a.name.cmp(&b.name))
544            // Use additional fields as deterministic tie-breakers.
545            .then_with(|| a.host.cmp(&b.host))
546            .then_with(|| a.value.cmp(&b.value))
547            .then_with(|| a.expiry.cmp(&b.expiry))
548            .then_with(|| a.secure.cmp(&b.secure))
549            .then_with(|| a.http_only.cmp(&b.http_only))
550    });
551}

553/// Applies include and exclude filters to the sorted cookie list.
554///
555/// Filtering uses only the cookie host.
556///
557/// The logic is:
558///
559/// ```text
560/// if host matches any exclude:
561///     remove
562/// else if there are no includes:
563///     keep
564/// else if host matches any include:
565///     keep
566/// else:
567///     remove
568/// ```
569///
570/// Include and exclude matching is case-insensitive and uses substring
571/// matching, including any leading dot in the host or filter.
572fn filter_cookies(
573    cookies: &mut Vec<Cookie>,
574    includes: &[String],
575    excludes: &[String],
576) {
577    // Lowercase all filters once instead of once per cookie.
578    let includes: Vec<String> =
579        includes.iter().map(|value| value.to_lowercase()).collect();
580
581    // Lowercase all exclusion filters once.
582    let excludes: Vec<String> =
583        excludes.iter().map(|value| value.to_lowercase()).collect();
584
585    // Retain only cookies that pass the filter rules.
586    cookies.retain(|cookie| {
587        // Lowercase the host for case-insensitive matching.
588        let host = cookie.host.to_lowercase();
589
590        // Any exclude match removes the cookie.
591        let excluded =
592            excludes.iter().any(|pattern| host.contains(pattern));
593
594        // Exclusion always takes precedence.
595        if excluded {
596            return false;
597        }
598
599        // With no include filters, retain every non-excluded cookie.
600        if includes.is_empty() {
601            return true;
602        }
603
604        // With includes, at least one include must match.
605        includes.iter().any(|pattern| host.contains(pattern))
606    });
607}

609/// Serializes cookies into a cached Netscape-format byte buffer.
610fn serialize_netscape(cookies: &[Cookie]) -> Vec<u8> {
611    // Start with an empty output buffer.
612    let mut output = Vec::new();
613
614    // Write explanatory comment lines.
615    output.extend_from_slice(b"# Netscape HTTP Cookie File\n");
616    output.extend_from_slice(
617        format!(
618            "# Generated by {} v{}\n",
619            env!("CARGO_PKG_NAME"),
620            env!("CARGO_PKG_VERSION"),
621        )
622            .as_bytes(),
623    );
624    output.extend_from_slice(
625        b"# Fields: domain<TAB>include_subdomains<TAB>path<TAB>secure<TAB>expiration<TAB>name<TAB>value\n",
626    );
627    output.extend_from_slice(
628        b"# expiration is a Unix timestamp in seconds\n",
629    );
630    output.extend_from_slice(
631        b"# include_subdomains is TRUE when the domain begins with a dot\n",
632    );
633    output.extend_from_slice(b"#\n");
634
635    // Serialize each sorted and filtered cookie.
636    for cookie in cookies {
637        // Netscape represents HttpOnly using a domain prefix.
638        let domain = if cookie.http_only {
639            format!("#HttpOnly_{}", cookie.host)
640        } else {
641            cookie.host.clone()
642        };
643
644        // Field 1: domain.
645        append_field(&mut output, &domain);
646        output.push(b'\t');
647
648        // Field 2: whether the domain applies to subdomains.
649        append_field(
650            &mut output,
651            if cookie.host.starts_with('.') {
652                "TRUE"
653            } else {
654                "FALSE"
655            },
656        );
657        output.push(b'\t');
658
659        // Field 3: cookie path.
660        append_field(&mut output, &cookie.path);
661        output.push(b'\t');
662
663        // Field 4: Secure flag.
664        append_field(
665            &mut output,
666            if cookie.secure { "TRUE" } else { "FALSE" },
667        );
668        output.push(b'\t');
669
670        // Field 5: Unix expiration timestamp.
671        append_field(&mut output, &cookie.expiry.to_string());
672        output.push(b'\t');
673
674        // Field 6: cookie name.
675        append_field(&mut output, &cookie.name);
676        output.push(b'\t');
677
678        // Field 7: cookie value.
679        append_field(&mut output, &cookie.value);
680        output.push(b'\n');
681    }
682
683    // Return the complete cached byte buffer.
684    output
685}

687/// Appends one Netscape field to the output buffer.
688///
689/// Tabs and newlines are percent-encoded so they cannot create extra
690/// columns or records in the tab-separated output format.
691fn append_field(
692    output: &mut Vec<u8>,
693    value: &str,
694) {
695    // Process the value as UTF-8 bytes.
696    for byte in value.bytes() {
697        match byte {
698            // Encode tabs as `%09`.
699            b'\t' => output.extend_from_slice(b"%09"),
700
701            // Encode line feeds as `%0A`.
702            b'\n' => output.extend_from_slice(b"%0A"),
703
704            // Encode carriage returns as `%0D`.
705            b'\r' => output.extend_from_slice(b"%0D"),
706
707            // Copy all other bytes unchanged.
708            _ => output.push(byte),
709        }
710    }
