156async fn main() -> Result<()> {
157    let args = Args::parse();
158    if !args.path.is_file() {
159        eprintln!(
160            "Error: input path is missing or is not a regular file: {}",
161            args.path.display()
162        );
163        eprintln!("Exit code: 4");
164        std::process::exit(4);
165    }
166    let filters = NormalizedFilters::from_args(&args);
167    print_arguments(&args, &filters);
168    let input_type = detect_input_type(&args.path)?;
169    let mut cookies = match input_type {
170        InputType::Sqlite => load_sqlite_cookies(&args.path).await?,
171        InputType::Netscape => load_netscape_cookies(&args.path)?,
172    };
173    eprintln!("Loaded cookies: {}", cookies.len());
174    apply_normal_filters(&mut cookies, &args, &filters);
175    eprintln!(
176        "After all include/exclude filters: {}",
177        cookies.len()
178    );
179    if !args.exclude_expired.is_empty() {
180        let references = args.exclude_expired.iter().map(|value| {
181            parse_expiration_reference(value.as_deref()).unwrap_or_else(|error| {
182                let displayed = value.as_deref().unwrap_or("expired");
183                eprintln!("Error: invalid --exclude-expired value: {displayed}");
184                eprintln!("Reason: {error}");
185                eprintln!("Exit code: 3");
186                std::process::exit(3);
187            })
188        }).collect::<Vec<_>>();
189        let selected = references
190            .iter()
191            .copied()
192            .min_by_key(|reference| reference.epoch_seconds)
193            .expect("references cannot be empty");
194        let (expired_count, session_count) =
195            apply_expiration_filter(&mut cookies, selected);
196        eprintln!(
197            "Expiration filters supplied: {}",
198            references.len()
199        );
200        eprintln!(
201            "Selected oldest expiration reference: {} UTC",
202            format_utc(selected.epoch_seconds)
203        );
204        eprintln!("Expired cookies excluded: {expired_count}");
205        eprintln!("Session cookies excluded: {session_count}");
206        eprintln!("After expiration filter: {}", cookies.len());
207    }
208    sort_cookies(&mut cookies);
209    let output = serialize_netscape(&cookies);
210    let stdout = io::stdout();
211    let mut writer = BufWriter::new(stdout.lock());
212    writer.write_all(&output).context("failed to write output")?;
213    writer.flush().context("failed to flush output")?;
214    Ok(())
215}

264fn detect_input_type(path: &Path) -> Result<InputType> {
265    let mut file = File::open(path).with_context(|| {
266        format!("failed to open {}", path.display())
267    })?;
268    let mut header = [0u8; 16];
269    let bytes_read = file.read(&mut header).with_context(|| {
270        format!("failed to read {}", path.display())
271    })?;
272    if bytes_read == 16 && &header == b"SQLite format 3\0" {
273        Ok(InputType::Sqlite)
274    } else {
275        Ok(InputType::Netscape)
276    }

278async fn load_sqlite_cookies(path: &Path) -> Result<Vec<Cookie>> {
279    let row_count = {
280        let path = path.to_path_buf();
281        tokio::task::spawn_blocking(move || count_rows(&path))
282            .await
283            .context("row-count task panicked")??
284    };
285    if row_count == 0 {
286        return Ok(Vec::new());
287    }
288    let available_workers = std::thread::available_parallelism()
289        .map(|value| value.get())
290        .unwrap_or(1);
291    let worker_count = available_workers
292        .saturating_sub(1)
293        .clamp(1, 8)
294        .min(row_count);
295    let rows_per_worker = row_count.div_ceil(worker_count);
296    eprintln!("SQLite rows: {row_count}");
297    eprintln!("SQLite workers: {worker_count}");
298    let database_path = Arc::new(path.to_path_buf());
299    let mut handles: Vec<JoinHandle<Result<Vec<Cookie>>>> =
300        Vec::with_capacity(worker_count);
301    for worker_index in 0..worker_count {
302        let offset = worker_index * rows_per_worker;
303        let remaining = row_count.saturating_sub(offset);
304        let limit = remaining.min(rows_per_worker);
305        if limit == 0 {
306            continue;
307        }
308        let database_path = Arc::clone(&database_path);
309        let handle = tokio::task::spawn_blocking(move || {
310            read_sqlite_chunk(&database_path, offset, limit)
311        });
312        handles.push(handle);
313    }
314    let mut cookies = Vec::with_capacity(row_count);
315    for handle in handles {
316        let worker_cookies = handle
317            .await
318            .context("SQLite reader task panicked")??;
319        cookies.extend(worker_cookies);
320    }
321    Ok(cookies)
322}

330fn count_rows(path: &Path) -> Result<usize> {
331    let connection = open_readonly(path)?;
332    let count: i64 = connection.query_row(
333        "SELECT COUNT(*) FROM moz_cookies",
334        [],
335        |row| row.get(0),
336    )?;
337    usize::try_from(count)
338        .context("SQLite row count does not fit in usize")
339}
340fn read_sqlite_chunk(
341    path: &Path,
342    offset: usize,
343    limit: usize,
344) -> Result<Vec<Cookie>> {
345    let connection = open_readonly(path)?;
346    let mut statement = connection.prepare(
347        r#"
348        SELECT
349            host,
350            path,
351            isSecure,
352            isHttpOnly,
353            expiry,
354            name,
355            value
356        FROM moz_cookies
357        LIMIT ?1 OFFSET ?2
358        "#,
359    )?;
360    let limit_i64 = i64::try_from(limit)
361        .context("chunk size does not fit in SQLite integer")?;
362    let offset_i64 = i64::try_from(offset)
363        .context("offset does not fit in SQLite integer")?;
364    let mut rows = statement.query((limit_i64, offset_i64))?;
365    let mut cookies = Vec::with_capacity(limit);
366    while let Some(row) = rows.next()? {
367        let host =
368            row.get::<_, Option<String>>(0)?.unwrap_or_default();
369        let path = row
370            .get::<_, Option<String>>(1)?
371            .unwrap_or_else(|| "/".to_string());
372        let is_secure = row.get::<_, Option<i64>>(2)?.unwrap_or(0);
373        let is_http_only =
374            row.get::<_, Option<i64>>(3)?.unwrap_or(0);
375        let expiry = row.get::<_, Option<i64>>(4)?.unwrap_or(0);
376        let name =
377            row.get::<_, Option<String>>(5)?.unwrap_or_default();
378        let value =
379            row.get::<_, Option<String>>(6)?.unwrap_or_default();
380        cookies.push(Cookie {
381            host,
382            path,
383            secure: is_secure != 0,
384            http_only: is_http_only != 0,
385            expiry,
386            name,
387            value,
388        });
389    }
390    Ok(cookies)
391}
392fn load_netscape_cookies(path: &Path) -> Result<Vec<Cookie>> {
393    let file = File::open(path).with_context(|| {
394        format!("failed to open {}", path.display())
395    })?;
396    let reader = BufReader::new(file);
397    let mut cookies = Vec::new();
398    for (line_index, line_result) in reader.lines().enumerate() {
399        let line_number = line_index + 1;
400        let line = line_result.with_context(|| {
401            format!("failed reading line {line_number}")
402        })?;
403        let line = line.trim_end_matches('\r');
404        if line.is_empty() {
405            continue;
406        }
407        if line.starts_with('#') && !line.starts_with("#HttpOnly_") {
408            continue;
409        }
410        let fields: Vec<&str> = line.split('\t').collect();
411        if fields.len() != 7 {
412            anyhow::bail!(
413                "invalid Netscape cookie line {line_number}: \
414                 expected 7 tab-separated fields, found {}",
415                fields.len()
416            );
417        }
418        let (http_only, host) = if let Some(host) =
419            fields[0].strip_prefix("#HttpOnly_")
420        {
421            (true, host.to_string())
422        } else {
423            (false, fields[0].to_string())
424        };
425        parse_bool_field(fields[1]).with_context(|| { format!("invalid include_subdomains field on line {line_number}") })?;
426        let secure =
427            parse_bool_field(fields[3]).with_context(|| {
428                format!("invalid secure field on line {line_number}")
429            })?;
430        let expiry =
431            fields[4].parse::<i64>().with_context(|| {
432                format!(
433                    "invalid expiration field on line {line_number}"
434                )
435            })?;
436        cookies.push(Cookie {
437            host,
438            path: fields[2].to_string(),
439            secure,
440            http_only,
441            expiry,
442            name: fields[5].to_string(),
443            value: fields[6].to_string(),
444        });
445    }
446    Ok(cookies)
447}
448fn parse_bool_field(value: &str) -> Result<bool> {
449    match value.trim().to_ascii_lowercase().as_str() {
450        "true" | "1" => Ok(true),
451        "false" | "0" => Ok(false),
452        other => anyhow::bail!("expected TRUE/FALSE, got {other:?}"),
453    }
454}
455fn apply_normal_filters(
456    cookies: &mut Vec<Cookie>,
457    args: &Args,
458    filters: &NormalizedFilters,
459) {
460    let has_include_filters = !filters.include_hosts.is_empty()
461        || !filters.include_paths.is_empty()
462        || !filters.include_cookie_names.is_empty()
463        || !filters.include_cookie_values.is_empty()
464        || args.include_is_subdomain
465        || args.include_is_secured;
466    if has_include_filters {
467        cookies.retain(|cookie| {
468            let host = cookie.host.to_ascii_lowercase();
469            let path = cookie.path.to_ascii_lowercase();
470            let name = cookie.name.to_ascii_lowercase();
471            let value = cookie.value.to_ascii_lowercase();
472            filters
473                .include_hosts
474                .iter()
475                .any(|pattern| host.contains(pattern))
476                || filters
477                .include_paths
478                .iter()
479                .any(|pattern| path.contains(pattern))
480                || filters
481                .include_cookie_names
482                .iter()
483                .any(|pattern| name.contains(pattern))
484                || filters
485                .include_cookie_values
486                .iter()
487                .any(|pattern| value.contains(pattern))
488                || (args.include_is_subdomain
489                && cookie.host.starts_with('.'))
490                || (args.include_is_secured && cookie.secure)
491        });
492    }
493    cookies.retain(|cookie| {
494        let host = cookie.host.to_ascii_lowercase();
495        let path = cookie.path.to_ascii_lowercase();
496        let name = cookie.name.to_ascii_lowercase();
497        let value = cookie.value.to_ascii_lowercase();
498        let excluded = filters
499            .exclude_hosts
500            .iter()
501            .any(|pattern| host.contains(pattern))
502            || filters
503            .exclude_paths
504            .iter()
505            .any(|pattern| path.contains(pattern))
506            || filters
507            .exclude_cookie_names
508            .iter()
509            .any(|pattern| name.contains(pattern))
510            || filters
511            .exclude_cookie_values
512            .iter()
513            .any(|pattern| value.contains(pattern))
514            || (args.exclude_is_subdomain
515            && cookie.host.starts_with('.'))
516            || (args.exclude_is_secured && cookie.secure);
517        !excluded
518    });
519}
520fn apply_expiration_filter(
521    cookies: &mut Vec<Cookie>,
522    reference: ExpirationReference,
523) -> (usize, usize) {
524    let mut expired_count = 0;
525    let mut session_count = 0;
526    cookies.retain(|cookie| {
527        let expired = cookie.expiry <= reference.epoch_seconds;
528        if expired {
529            expired_count += 1;
530            if cookie.expiry == 0 {
531                session_count += 1;
532            }
533        }
534        !expired
535    });
536    (expired_count, session_count)
537}
538fn parse_expiration_reference(
539    value: Option<&str>
540) -> Result<ExpirationReference, ExpirationParseError> {
541    let value = value.unwrap_or("expired").trim();
542    if value.eq_ignore_ascii_case("expired") {
543        return Ok(ExpirationReference {
544            epoch_seconds: Utc::now().timestamp(),
545        });
546    }
547    if value.len() >= 4 && value[..4].eq_ignore_ascii_case("date") {
548        return parse_date_reference(&value[4..]);
549    }
550    if value.len() >= 5 && value[..5].eq_ignore_ascii_case("epoch") {
551        return parse_epoch_reference(&value[5..]);
552    }
553    Err(ExpirationParseError::InvalidFormat)
554}
555fn parse_date_reference(
556    value: &str
557) -> Result<ExpirationReference, ExpirationParseError> {
558    let digits: String = value
559        .chars()
560        .take_while(|character| character.is_ascii_digit())
561        .collect();
562    if !matches!(digits.len(), 4 | 6 | 8 | 10 | 12 | 14) {
563        return Err(ExpirationParseError::InvalidDate);
564    }
565    let year = digits[0..4]
566        .parse::<i32>()
567        .map_err(|_| ExpirationParseError::InvalidDate)?;
568    let month = if digits.len() >= 6 {
569        digits[4..6]
570            .parse::<u32>()
571            .map_err(|_| ExpirationParseError::InvalidDate)?
572    } else {
573        1
574    };
575    let day = if digits.len() >= 8 {
576        digits[6..8]
577            .parse::<u32>()
578            .map_err(|_| ExpirationParseError::InvalidDate)?
579    } else {
580        1
581    };
582    let hour = if digits.len() >= 10 {
583        digits[8..10]
584            .parse::<u32>()
585            .map_err(|_| ExpirationParseError::InvalidDate)?
586    } else {
587        0
588    };
589    let minute = if digits.len() >= 12 {
590        digits[10..12]
591            .parse::<u32>()
592            .map_err(|_| ExpirationParseError::InvalidDate)?
593    } else {
594        0
595    };
596    let second = if digits.len() >= 14 {
597        digits[12..14]
598            .parse::<u32>()
599            .map_err(|_| ExpirationParseError::InvalidDate)?
600    } else {
601        0
602    };
603    let date = NaiveDate::from_ymd_opt(year, month, day)
604        .ok_or(ExpirationParseError::InvalidDate)?;
605    let datetime = date
606        .and_hms_opt(hour, minute, second)
607        .ok_or(ExpirationParseError::InvalidDate)?;
608    Ok(ExpirationReference {
609        epoch_seconds: datetime.and_utc().timestamp(),
610    })
611}
612fn parse_epoch_reference(
613    value: &str
614) -> Result<ExpirationReference, ExpirationParseError> {
615    let digits: String = value
616        .chars()
617        .take_while(|character| character.is_ascii_digit())
618        .collect();
619    if digits.is_empty() {
620        return Err(ExpirationParseError::InvalidEpoch);
621    }
622    let normalized = if digits.len() < 10 {
623        format!("{digits:0>10}")
624    } else {
625        digits[..10].to_string()
626    };
627    let epoch_seconds = normalized
628        .parse::<i64>()
629        .map_err(|_| ExpirationParseError::InvalidEpoch)?;
630    Ok(ExpirationReference { epoch_seconds })
631}
632fn format_utc(epoch_seconds: i64) -> String {
633    DateTime::<Utc>::from_timestamp(epoch_seconds, 0)
634        .map(|datetime| {
635            datetime.format("%Y-%m-%dT%H:%M:%SZ").to_string()
636        })
637        .unwrap_or_else(|| "invalid UTC timestamp".to_string())
638}
639fn sort_cookies(cookies: &mut [Cookie]) {
640    cookies.sort_unstable_by(|a, b| {
641        let a_host = a.host.trim_start_matches('.');
642        let b_host = b.host.trim_start_matches('.');
643        a_host
644            .cmp(b_host)
645            .then_with(|| a.path.cmp(&b.path))
646            .then_with(|| a.name.cmp(&b.name))
647            .then_with(|| a.host.cmp(&b.host))
648            .then_with(|| a.value.cmp(&b.value))
649            .then_with(|| a.expiry.cmp(&b.expiry))
650            .then_with(|| a.secure.cmp(&b.secure))
651            .then_with(|| a.http_only.cmp(&b.http_only))
652    });
653}
654fn serialize_netscape(cookies: &[Cookie]) -> Vec<u8> {
655    let mut output = Vec::new();
656    output.extend_from_slice(b"# Netscape HTTP Cookie File\n");
657    output.extend_from_slice(
658        format!(
659            "# Generated by {} v{}\n",
660            env!("CARGO_PKG_NAME"),
661            env!("CARGO_PKG_VERSION")
662        )
663            .as_bytes(),
664    );
665    output.extend_from_slice(b"# Fields: domain<TAB>include_subdomains<TAB>path<TAB>secure<TAB>\
666expiration<TAB>name<TAB>value\n", );
667    output.extend_from_slice(
668        b"# expiration is a Unix timestamp in seconds\n",
669    );
670    output.extend_from_slice(b"# include_subdomains is TRUE when the domain begins with a dot\n");
671    output.extend_from_slice(b"#\n");
672    for cookie in cookies {
673        let domain = if cookie.http_only {
674            format!("#HttpOnly_{}", cookie.host)
675        } else {
676            cookie.host.clone()
677        };
678        append_field(&mut output, &domain);
679        output.push(b'\t');
680        append_field(
681            &mut output,
682            if cookie.host.starts_with('.') {
683                "TRUE"
684            } else {
685                "FALSE"
686            },
687        );
688        output.push(b'\t');
689        append_field(&mut output, &cookie.path);
690        output.push(b'\t');
691        append_field(
692            &mut output,
693            if cookie.secure { "TRUE" } else { "FALSE" },
694        );
695        output.push(b'\t');
696        append_field(&mut output, &cookie.expiry.to_string());
697        output.push(b'\t');
698        append_field(&mut output, &cookie.name);
699        output.push(b'\t');
700        append_field(&mut output, &cookie.value);
701        output.push(b'\n');
702    }
703    output
704}
705fn append_field(
706    output: &mut Vec<u8>,
707    value: &str,
708) {
709    for byte in value.bytes() {
710        match byte {
711            b'\t' => output.extend_from_slice(b"%09"),
712            b'\n' => output.extend_from_slice(b"%0A"),
713            b'\r' => output.extend_from_slice(b"%0D"),
714            _ => output.push(byte),
715        }
716    }
717}
